DETERMINAR QUAL ABA CARREGAR if (paramCliente === "") { /* MODO ACESSO DIRETO: URL sem parâmetro. Força o carregamento do Slot_0. */ gidParaCarregar = GID_PRINCIPAL; modoAcessoDireto = true; console.log("Acesso Direto Detectado. Carregando Slot_0 (GID Principal)."); } else { /* MODO ACESSO INDIVIDUAL: Busca o slot correspondente na lista. */ const slotMap = slotsIndividuais.find(s => s.id === paramCliente); if (slotMap) { gidParaCarregar = slotMap.gid; console.log(`Acesso Individual Detectado para ${paramCliente}. Carregando GID ${gidParaCarregar}.`); } else { console.log(`Slot ${paramCliente} não encontrado na lista de acessos individuais.`); } } // Se não determinamos um GID válido (parâmetro inválido), exibe erro if (gidParaCarregar === "") { exibirErro("Link inválido ou desativado."); return; } // 2. BUSCAR DADOS NA PLANILHA const csvUrl = `https://docs.google.com/spreadsheets/d/e/2PACX-1vTyTRyEdutIaD7zX3fwiUxOu_j_U4yzO2OwfaI0Oy_1Keum9q2oHSxSGbsbSzcRuWgrqFaWWsw4PIRJ/pub?gid=${gidParaCarregar}&single=true&output=csv&t=${new Date().getTime()}`; try { const response = await fetch(csvUrl); const text = await response.text(); const rows = parseCSV(text); function getCellVal(r, c) { return (rows[r] && rows[r][c]) ? rows[r][c].trim() : ""; } const clientePlanilha = getCellVal(2, 1); const clienteNorm = normalizarTexto(clientePlanilha); // 3. VALIDAR DADOS let validacaoSucesso = false; if (modoAcessoDireto) { /* No modo acesso direto ao Slot_0, não validamos nome, apenas carregamos. */ validacaoSucesso = true; } else { /* No modo individual, validamos se o nome na aba bate com o parâmetro da busca. */ if (clienteNorm.includes(paramCliente) || paramCliente.includes(clienteNorm)) { validacaoSucesso = true; } } if (validacaoSucesso) { dadosEncontrados = { cliente: clientePlanilha, orcamento: getCellVal(3, 1), criterios: getCellVal(4, 1), etapa: getCellVal(5, 1), vistoria: getCellVal(6, 1), c1Name: getCellVal(2, 4), c1Status: getCellVal(2, 5), c2Name: getCellVal(3, 4), c2Status: getCellVal(3, 5), c3Name: getCellVal(4, 4), c3Status: getCellVal(4, 5), obs: getCellVal(18, 0) }; } } catch (err) { console.error("Erro ao carregar dados do GSheet:", err); } // 4. PREENCHER TELA OU EXIBIR ERRO if (dadosEncontrados) { preencherTelaComDados(dadosEncontrados); } else { exibirErro(modoAcessoDireto ? "Erro ao carregar os dados principais." : "Cliente Não Encontrado."); } } /* =========================================================== FUNÇÕES AUXILIARES MANTIDAS =========================================================== */ // Função para preencher os elementos HTML com os dados function preencherTelaComDados(dados) { document.getElementById('cliente-nome').innerText = dados.cliente || "-"; document.getElementById('orcamento-val').innerText = dados.orcamento || "-"; document.getElementById('criterios-val').innerText = dados.criterios || "-"; document.getElementById('vistoria-status').innerText = dados.vistoria || "Em andamento"; document.getElementById('obs-text').innerText = dados.obs || "A busca continua. Nossos especialistas estão monitorando o mercado."; let carsHtml = ''; if(dados.c1Name) carsHtml += `
${dados.c1Name}${formatCarBadge(dados.c1Status)}
`; if(dados.c2Name) carsHtml += `
${dados.c2Name}${formatCarBadge(dados.c2Status)}
`; if(dados.c3Name) carsHtml += `
${dados.c3Name}${formatCarBadge(dados.c3Status)}
`; document.getElementById('cars-list').innerHTML = carsHtml || '
Buscando as melhores opções no mercado...
'; const currentStep = parseInt(String(dados.etapa).charAt(0)) || 1; const pct = (currentStep / 5) * 100; document.getElementById('progresso-pct').innerText = pct + '%'; setTimeout(() => { document.getElementById('progresso-bar').style.width = pct + '%'; }, 150); for (let i = 1; i <= 5; i++) { const el = document.getElementById('step-' + i); if(el) { el.classList.remove('completed', 'active'); if (i < currentStep) el.classList.add('completed'); else if (i === currentStep) el.classList.add('active'); } } } // Função para exibir mensagem de erro na tela function exibirErro(mensagem) { document.getElementById('cliente-nome').innerText = "Erro"; document.getElementById('orcamento-val').innerText = "-"; document.getElementById('criterios-val').innerText = "-"; document.getElementById('vistoria-status').style.display = 'none'; document.getElementById('cars-list').innerHTML = `
${mensagem}
`; document.getElementById('obs-text').innerText = "Por favor, verifique o link ou tente novamente mais tarde."; document.getElementById('progresso-pct').innerText = '0%'; document.getElementById('progresso-bar').style.width = '0%'; }
